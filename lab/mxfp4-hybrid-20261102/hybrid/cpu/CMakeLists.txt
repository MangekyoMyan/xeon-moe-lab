cmake_minimum_required(VERSION 3.14)
project(deepseek_cpu_review LANGUAGES C CXX)
set(REVIEW_PROFILE "portable" CACHE STRING "portable, ivybridge or native")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

foreach(feature AVX AVX2 AVX_VNNI AVX512 AVX512_VBMI AVX512_VNNI AVX512_BF16 FMA F16C SSE42 BMI2 AMX_TILE AMX_INT8 AMX_BF16)
    set(GGML_${feature} OFF CACHE BOOL "Explicit review ISA" FORCE)
endforeach()
set(GGML_NATIVE OFF CACHE BOOL "No implicit host ISA" FORCE)
if(REVIEW_PROFILE STREQUAL "ivybridge")
    if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
        message(FATAL_ERROR "Ivy Bridge profile requires x86_64")
    endif()
    set(GGML_AVX ON CACHE BOOL "" FORCE)
    set(GGML_F16C ON CACHE BOOL "" FORCE)
    set(GGML_SSE42 ON CACHE BOOL "" FORCE)
    add_compile_options(-march=ivybridge -mtune=ivybridge -mno-avx2 -mno-fma)
elseif(REVIEW_PROFILE STREQUAL "native")
    set(GGML_NATIVE ON CACHE BOOL "" FORCE)
elseif(NOT REVIEW_PROFILE STREQUAL "portable")
    message(FATAL_ERROR "Unknown REVIEW_PROFILE")
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
    add_compile_options(-march=x86-64 -mno-avx -mno-avx2 -mno-fma -mno-f16c)
endif()

foreach(backend CUDA HIP VULKAN SYCL OPENCL CANN MUSA METAL RPC WEBGPU OPENVINO ZENDNN ZDNN BLAS CPU_KLEIDIAI CPU_HBM)
    set(GGML_${backend} OFF CACHE BOOL "CPU-only offline review" FORCE)
endforeach()
set(GGML_CPU ON CACHE BOOL "" FORCE)
set(GGML_CPU_REPACK OFF CACHE BOOL "Matches current engine layout" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_USE_PREBUILT_UI OFF CACHE BOOL "No network downloads" FORCE)
set(LLAMA_CONNECT OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_LLGUIDANCE OFF CACHE BOOL "" FORCE)
set(LLAMA_TOOLS_INSTALL OFF CACHE BOOL "" FORCE)
set(LLAMA_TESTS_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../engine" engine)
add_executable(cpu-quant-probe quant_probe.cpp)
target_compile_features(cpu-quant-probe PRIVATE cxx_std_17)
target_link_libraries(cpu-quant-probe PRIVATE ggml-cpu ggml-base)
