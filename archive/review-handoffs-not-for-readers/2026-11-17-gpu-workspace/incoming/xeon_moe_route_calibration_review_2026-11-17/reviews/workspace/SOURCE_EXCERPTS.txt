   31 // returns the suggested maximum number of rows to process at once, given the temporary buffer bytes per row
   32 int ggml_cuda_chunk_nrows(const size_t row_bytes, const int64_t nrows) {
   33     // process rows in chunks up to approximately this size (currently 64MB)
   34     // to avoid excessive temporary buffers memory usage
   35     const int chunk_bytes = 1 << 26;
   36 
   37     // calculate how many rows will fit in one chunk (must be at least one)
   38     const int chunk_nrows = std::max((int) (chunk_bytes / row_bytes), 1);
   39 
   40     // limit the resulting amount to total nrows
   41     return std::min((int64_t) chunk_nrows, nrows);
